CREATE TABLE word_derivatives (
    word_id INT NOT NULL,
    derivative_id INT NOT NULL,
    PRIMARY KEY (word_id, derivative_id),
    FOREIGN KEY (word_id) REFERENCES words(id) ON DELETE CASCADE,
    FOREIGN KEY (derivative_id) REFERENCES words(id) ON DELETE CASCADE
);

-- Crucial index for the reverse lookup (finding the base word from a derivative)
CREATE INDEX idx_derivative_to_word ON word_derivatives(derivative_id, word_id);

export const getDerivatives = async (req: Request, res: Response) => {
  const { word } = req.params;
  
  const query = `
    SELECT w2.word 
    FROM words w1
    JOIN word_derivatives wd ON w1.id = wd.word_id
    JOIN words w2 ON wd.derivative_id = w2.id
    WHERE w1.word = ?
  `;

  try {
    const [rows] = await db.execute(query, [word]);
    res.json({ word, derivatives: (rows as any[]).map(r => r.word) });
  } catch (error) {
    res.status(500).json({ error: 'Database query failed' });
  }
};
